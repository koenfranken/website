(() => {
  // Openingstijden per dag (0 = zondag): [van, tot] in uren
  const tijden = { 1: [9, 17], 2: [9, 17], 3: [9, 13], 4: [9, 17], 5: [9, 15] };

  const nu = new Date(), dag = nu.getDay(), uur = nu.getHours() + nu.getMinutes() / 60;

  const open = vandaag && uur >= vandaag[0] && uur < vandaag[1];
